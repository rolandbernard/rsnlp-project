  \begin{items}
    \begin{oitem}{\litem}
      Self-Supervised Learning \\
      \sitem {\footnotesize Learning from the data itself without explicit human-provided labels.} \\
      \sitem {\footnotesize Define some pretext task based on the input data.} \\
      \sitem {\footnotesize Has been effective in other domain like CV and NLP.}
    \end{oitem}
    \begin{oitem}{\litem}
      Contrastive Learning \\
      \sitem {\footnotesize The pretext task is to compare different instances.} \\
      \sitem {\footnotesize The model learns to tell similar things apart from dissimilar things.} \\

      \sitem {\footnotesize Negative pair: Augmented versions of different inputs.}

\begin{frame}
\frametitle{CL4SRec --- Model Architecture}
  \begin{items}
    \begin{oitem}{\litem}
      User Representation Model based on Transformer (from SASRec) \\
      \sitem {\footnotesize Items and position embedding layer.} \\
      \sitem {\footnotesize Multi-head self-attention} \\
      \sitem {\footnotesize Position-wise feed-forward network} \\
      \sitem {\footnotesize Layer normalization and dropout for regularization}
    \end{oitem}
    \begin{oitem}{\litem}
      Next Item Probability Prediction \\
      \sitem {\footnotesize Based on dot product between user and item representation.} \\
      \sitem {\footnotesize The items closest to the user representation are predicted.}
    \end{oitem}
  \end{items}
\end{frame}

\begin{frame}
\frametitle{CL4SRec --- Training Process}
  \begin{items}
    \begin{oitem}{\litem}
      Recommendation Loss \\
      \sitem {\footnotesize Typically cross-entropy, predicting the true next item versus other items.} \\
      \sitem {\footnotesize Uses only a sampled softmax for efficiency.} \\
      \sitem {\footnotesize $\mathcal{L}_\mathrm{main}(s_u) = - \log \frac{\exp \left( s_u \cdot v^+ \right)}{\exp \left( s_u \cdot v^+ \right) + \sum_{v^- \in V^-} \exp \left( s_u \cdot v^- \right)}$}
    \end{oitem}
    \begin{oitem}{\litem}
      Contrastive Loss \\
      \sitem {\footnotesize Aims to maximize similarity for positive pairs and minimize for negative pairs.} \\
      \sitem {\footnotesize $\mathcal{L}_\mathrm{cl}\left( s^{a_i}_u, s^{a_j}_u \right) = - \log \frac{\exp \left( s^{a_i}_u \cdot s^{a_j}_u \right)}{\exp \left( s^{a_i}_u \cdot s^{a_j}_u \right) + \sum_{s^- \in S^-} \exp \left( s^{a_i}_u \cdot s^- \right)}$}
    \end{oitem}
    \begin{oitem}{\litem}
      Multi-Task Learning \\
      \sitem {\footnotesize The model is trained to optimize both tasks simultaneously.} \\
      \sitem {\footnotesize $\mathcal{L}_\mathrm{total} = \mathcal{L}_\mathrm{main} + \lambda \mathcal{L}_\mathrm{cl}$}
    \end{oitem}
  \end{items}
\end{frame}
